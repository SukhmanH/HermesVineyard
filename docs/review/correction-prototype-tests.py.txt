"""Correction regressions using fixture records, never production data."""
from test_compliance import _commit_spray, _commit_task, _ready_draft

from vineyard_mcp.compliance import draft_correction, draft_task_log


def test_task_correction_validates_hours_and_keeps_workers(db, juan):
    first = draft_task_log(db, juan, {"task_type": "riego", "hours_total": 6})
    row = _commit_task(db, first["confirm_token"])["committed"]
    bad = draft_correction(db, "task_log", row["id"], {"wa_phone": juan, "hours_total": -6})
    assert not bad["ready_to_confirm"]
    assert "error" in _commit_task(db, bad["confirm_token"])
    good = draft_correction(db, "task_log", row["id"],
                            {"wa_phone": juan, "hours_total": 8, "notes": "corrected"})
    new = _commit_task(db, good["confirm_token"])["committed"]
    assert new["id"] != row["id"]
    assert new["corrects_log_id"] == row["id"]
    assert new["hours_total"] == 8
    assert db.execute("SELECT hours_total FROM task_log WHERE id=?",
                      (row["id"],)).fetchone()[0] == 6
    assert [r[0] for r in db.execute("SELECT id FROM task_log_current")] == [new["id"]]
    old_workers = list(db.execute("SELECT contact_id FROM task_workers WHERE task_log_id=?",
                                  (row["id"],)))
    new_workers = list(db.execute("SELECT contact_id FROM task_workers WHERE task_log_id=?",
                                  (new["id"],)))
    assert old_workers and [tuple(r) for r in old_workers] == [tuple(r) for r in new_workers]



def test_block_correction_changes_the_block(db, juan):
    original = _commit_spray(db, _ready_draft(db, juan)["confirm_token"])["committed"]
    corr = draft_correction(db, "spray_log", original["id"], {"block_code": "B1"})
    assert corr["draft"]["block_code"] == "B1"
    assert corr["draft"]["block_id"] != original["block_id"]


def test_corrected_rate_requires_new_label_check(db, juan):
    original = _commit_spray(db, _ready_draft(db, juan)["confirm_token"])["committed"]
    corr = draft_correction(db, "spray_log", original["id"], {"rate_value": 1000})
    assert not corr["ready_to_confirm"]
    assert "rate_confirmed" in {r["field"] for r in corr["missing_fields"]}


def test_two_prepared_corrections_cannot_both_supersede_one_row(db, juan):
    original = _commit_spray(db, _ready_draft(db, juan)["confirm_token"])["committed"]
    one = draft_correction(db, "spray_log", original["id"], {"notes": "first"})
    two = draft_correction(db, "spray_log", original["id"], {"notes": "second"})
    assert "committed" in _commit_spray(db, one["confirm_token"])
    assert _commit_spray(db, two["confirm_token"])["error"] == "already_superseded"
    assert db.execute("SELECT count(*) FROM spray_log_current").fetchone()[0] == 1


def test_correction_metadata_is_not_editable(db, juan):
    original = _commit_spray(db, _ready_draft(db, juan)["confirm_token"])["committed"]
    corr = draft_correction(db, "spray_log", original["id"], {"corrects_log_id": 999})
    assert "error" in corr



def test_product_correction_cannot_reuse_previous_products_label(db, juan):
    original = _commit_spray(db, _ready_draft(db, juan)["confirm_token"])["committed"]
    corr = draft_correction(db, "spray_log", original["id"],
                            {"product_name_raw": "Mystery Fungicide"})
    assert not corr["ready_to_confirm"]
    assert corr["draft"]["product_id"] != original["product_id"]
    assert corr["draft"].get("rei_hours") is None
    assert corr["draft"].get("pcp_number") is None
    assert "label_rei" in {r["field"] for r in corr["missing_fields"]}
    assert "error" in _commit_spray(db, corr["confirm_token"])
